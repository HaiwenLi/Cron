# OpenCode Cron M1(核心 runner + script/opencode 执行器 + sync + 命令)Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 交付 M1:无状态 Python runner(`run/list/validate/sync`)+ script/opencode 执行器 + 按任务频次的 schtasks 注册 + `/cron-list`,`/cron-run` 命令,可用 example-echo 任务端到端验收。

**Architecture:** 每任务一条 schtasks(按频次唤醒)→ `runner run <id>` 无状态进程:文件加载 → 无等待锁 → 活跃注册表并发闸 → 执行器适配(subprocess)→ runs/ 留痕退出。文件是唯一契约,命令文件只做展示与手动触发。

**Tech Stack:** Python 3.12(依赖 croniter、PyYAML、psutil;dev: pytest);Windows schtasks(XML 注册);opencode 命令 markdown。

**Spec:** `docs/spec/2026-10-26-opencode-cron-design.md`(本计划实现其 M1 子集,术语以 spec 为准)

## Global Constraints

- 测试一律从仓库根运行:`python -m pytest runner/tests -v`(全部离线,不真调 LLM、不真动 schtasks)
- 运行时依赖仅三个:croniter、PyYAML、psutil;禁引入调度框架(非目标:进程内调度)
- 入库内容零绝对路径、零个人信息;机器路径只允许出现在运行时生成物(state/、runs/、schtasks XML)与不入库文件(config.local.yaml)
- runner 无常驻:任何子命令执行完进程退出(G4)
- 同任务不重叠;全局并发闸 = 活跃权重和(spec §6.4);skipped 类退出码为 0(系统正常工作,任务被跳过),success=0,failed/timeout=1
- 所有 meta/锁/登记 JSON 字段名以本计划 Interfaces 为准,任务间不得私自改名
- Windows 优先;时钟、pid 存活、schtasks 后端必须可注入(测试不依赖真实系统状态)
- 每个任务一个 commit,message 注明测试结果(如 `(first execution: 6/6 PASS)`)

```python
# runner/cron_runner/config.py
from dataclasses import dataclass, field
from pathlib import Path
import re
import yaml

_VAR_RE = re.compile(r"\$\{([A-Za-z_][A-Za-z0-9_]*)\}")

class UndefinedVariableError(Exception):
    def __init__(self, name: str, text: str):
        super().__init__(f"undefined variable ${{{name}}} in: {text!r}")
        self.name = name

@dataclass
class Config:
    max_concurrent: int = 2
    retention_days: int = 14
    variables: dict = field(default_factory=dict)

def interpolate(text: str, variables: dict) -> str:
    def sub(m):
        name = m.group(1)
        if name not in variables:
            raise UndefinedVariableError(name, text)
        return variables[name]
    return _VAR_RE.sub(sub, text)

def _read(path: Path) -> dict:
    if not path.exists():
        return {}
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    if not isinstance(data, dict):
        raise ValueError(f"{path} must be a mapping")
    return data

def load_config(cron_home: Path) -> Config:
    base = _read(cron_home / "config.yaml")
    local = _read(cron_home / "config.local.yaml")
    merged = {**base, **local}
    variables = {"CRON_HOME": str(cron_home.resolve()), "PYTHON": "python"}
    variables.update(base.get("variables") or {})
    variables.update(local.get("variables") or {})
    return Config(
        max_concurrent=int(merged.get("max_concurrent", 2)),
        retention_days=int(merged.get("retention_days", 14)),
        variables=variables,
    )
```

```python
# runner/cron_runner/tasks.py
from dataclasses import dataclass, field
from pathlib import Path
import yaml
from croniter import croniter
from cron_runner.config import Config, interpolate, UndefinedVariableError

_EXECUTORS = {"script", "opencode", "kimi", "pi"}
_MISS = {"coalesce", "skip", "catchup"}
_PERM = {"default", "yolo", "auto"}
_TYPES = {"script", "prompt", "skill"}

class TaskValidationError(Exception):
    pass

@dataclass
class Step:
    name: str
    type: str
    run: str
    skill: str | None = None
    output: str | None = None

@dataclass
class Task:
    id: str
    name: str
    schedule: str
    executor: str
    steps: list
    enabled: bool = True
    miss_policy: str = "coalesce"
    permission: str = "auto"
    model: str | None = None
    timeout: int = 1800
    weight: int = 1
    workdir: str = "${WORKSPACE}"
    notify: tuple = ("toast",)

def _parse(path: Path) -> Task:
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    fname = path.stem
    if data.get("id") != fname:
        raise TaskValidationError(
            f"{path.name}: id {data.get('id')!r} must match filename {fname!r}")
    data.pop("id")
    steps = [Step(**s) for s in data.pop("steps")]
    return Task(id=fname, steps=steps, **data)

def load_tasks(cron_home: Path) -> list:
    by_id: dict = {}
    for rel in ("tasks.d", "tasks.d.local"):
        d = cron_home / rel
        if not d.exists():
            continue
        for p in sorted(d.glob("*.yaml")):
            by_id[p.stem] = _parse(p)
    return list(by_id.values())

def validate_task(task: Task, cfg: Config) -> list:
    errs = []
    if not croniter.is_valid(task.schedule):
        errs.append(f"schedule invalid: {task.schedule!r}")
    if task.executor not in _EXECUTORS:
        errs.append(f"executor must be one of {sorted(_EXECUTORS)}, got {task.executor!r}")
    if task.miss_policy not in _MISS:
        errs.append(f"miss_policy must be one of {sorted(_MISS)}")
    if task.permission not in _PERM:
        errs.append(f"permission must be one of {sorted(_PERM)}")
    if task.weight < 1:
        errs.append("weight must be >= 1")
    if task.timeout < 60:
        errs.append("timeout must be >= 60s")
    if len(task.steps) < 1:
        errs.append("at least one step required")
    elif len(task.steps) > 1:
        errs.append(f"multi-step steps arrive in M3, got {len(task.steps)}")
    else:
        st = task.steps[0]
        if st.type not in _TYPES:
            errs.append(f"step.type must be one of {sorted(_TYPES)}")
        if task.executor == "script" and st.type != "script":
            errs.append("executor 'script' requires step.type 'script'")
        if task.executor != "script" and st.type == "script":
            errs.append("agent executors require step.type prompt|skill")
    probe = dict(cfg.variables)
    probe.setdefault("WORKSPACE", "")
    probe["RUN_DIR"] = "<RUN_DIR>"
    for label, text in (("workdir", task.workdir), ("run", task.steps[0].run if task.steps else "")):
        try:
            interpolate(text, probe)
        except UndefinedVariableError as e:
            errs.append(f"{label}: {e}")
    return errs
```

```python
# runner/cron_runner/due.py
from datetime import datetime
from croniter import croniter

def next_run(schedule: str, after: datetime) -> datetime:
    it = croniter(schedule, after.replace(microsecond=0))
    return it.get_next(datetime)

def is_due(schedule: str, cursor: datetime | None, now: datetime) -> bool:
    if cursor is None:
        return True
    return next_run(schedule, cursor) <= now.replace(microsecond=0)
```

```python
import json, os, time, uuid
from dataclasses import dataclass
from pathlib import Path
import psutil

def boot_id() -> int:
    return int(psutil.boot_time())

def pid_alive(pid: int) -> bool:
    return psutil.pid_exists(pid)

@dataclass
class LockResult:
    acquired: bool
    reason: str = ""

def _read_json(path: Path):
    try:
        return json.loads(path.read_text(encoding="utf-8"))
    except Exception:
        return None

# 注意:所有注入参数用 None 默认值 + 函数体内晚绑定,
# 这样测试 monkeypatch locks.pid_alive / locks.boot_id 才生效
# (默认参数在 def 时求值,早绑定会吞掉 monkeypatch —— 已知陷阱)。
def acquire_task_lock(lock_path, task_id, timeout_sec, *, pid_checker=None,
                      boot_provider=None, now_provider=None, grace_sec=600):
    pid_checker = pid_checker or pid_alive
    boot_provider = boot_provider or boot_id
    now = now_provider() if now_provider else int(time.time())
    payload = {"pid": os.getpid(), "boot_id": boot_provider(),
               "task_id": task_id, "started_at": now}
    lock_path.parent.mkdir(parents=True, exist_ok=True)
    try:
        fd = os.open(lock_path, os.O_CREAT | os.O_EXCL | os.O_WRONLY)
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(payload, f)
        return LockResult(True)
    except FileExistsError:
        pass
    old = _read_json(lock_path)
    if old is None:
        reason = "corrupt"
    elif old.get("boot_id") != boot_provider():
        reason = "stale-boot"
    elif not pid_checker(old.get("pid", -1)):
        reason = "stale-dead-pid"
    elif now - old.get("started_at", 0) > timeout_sec + grace_sec:
        reason = "stale-age"
    else:
        return LockResult(False, "alive")
    tmp = lock_path.with_suffix(f".{uuid.uuid4().hex}.tmp")
    tmp.write_text(json.dumps(payload), encoding="utf-8")
    os.replace(tmp, lock_path)
    return LockResult(True, reason)

def register_active(active_dir, task_id, weight, *, pid=None, boot_provider=None,
                    now_provider=None):
    boot_provider = boot_provider or boot_id
    active_dir.mkdir(parents=True, exist_ok=True)
    payload = {"pid": os.getpid() if pid is None else pid,
               "boot_id": boot_provider(), "task_id": task_id,
               "weight": weight,
               "started_at": now_provider() if now_provider else int(time.time())}
    token = active_dir / f"{uuid.uuid4().hex}.json"
    token.write_text(json.dumps(payload), encoding="utf-8")
    return token

def active_weight(active_dir, *, pid_checker=None, boot_provider=None):
    pid_checker = pid_checker or pid_alive
    boot_provider = boot_provider or boot_id
    total = 0
    if not active_dir.exists():
        return 0
    for p in list(active_dir.glob("*.json")):
        data = _read_json(p)
        if data is None or data.get("boot_id") != boot_provider() \
                or not pid_checker(data.get("pid", -1)):
            p.unlink(missing_ok=True)
            continue
        total += int(data.get("weight", 1))
    return total

def unregister(token: Path) -> None:
    Path(token).unlink(missing_ok=True)
```

```python
# runner/cron_runner/history.py
import json, os
from datetime import datetime, timedelta
from pathlib import Path

def new_run_dir(runs_root: Path, task_id: str, now: datetime | None = None) -> Path:
    now = now or datetime.now()
    base = now.strftime("%Y%m%d-%H%M%S")
    d = runs_root / task_id
    d.mkdir(parents=True, exist_ok=True)
    cand = d / base
    n = 1
    while cand.exists():
        n += 1
        cand = d / f"{base}-{n}"
    cand.mkdir()
    return cand

def write_meta(run_dir: Path, meta: dict) -> None:
    meta.setdefault("finished_at", datetime.now().isoformat(timespec="seconds"))
    tmp = run_dir / "meta.json.tmp"
    tmp.write_text(json.dumps(meta, ensure_ascii=False, indent=2),
                   encoding="utf-8")
    os.replace(tmp, run_dir / "meta.json")

def _run_dirs(runs_root: Path, task_id: str) -> list:
    d = runs_root / task_id
    return sorted([p for p in d.iterdir() if p.is_dir()]) if d.exists() else []

def latest_run(runs_root: Path, task_id: str) -> dict | None:
    dirs = _run_dirs(runs_root, task_id)
    for p in reversed(dirs):
        f = p / "meta.json"
        if f.exists():
            try:
                return json.loads(f.read_text(encoding="utf-8"))
            except Exception:
                continue
    return None

def cleanup(runs_root: Path, retention_days: int, now: datetime | None = None):
    now = now or datetime.now()
    cutoff = now - timedelta(days=retention_days)
    deleted = []
    if not runs_root.exists():
        return deleted
    for task_dir in runs_root.iterdir():
        if not task_dir.is_dir():
            continue
        dirs = _run_dirs(runs_root, task_dir.name)
        for p in dirs[:-1]:  # 保留最新 1 个
            if datetime.fromtimestamp(p.stat().st_mtime) < cutoff:
                import shutil
                shutil.rmtree(p, ignore_errors=True)
                deleted.append(p)
    return deleted
```

```python
import subprocess, time
from dataclasses import dataclass, field
from pathlib import Path

@dataclass
class ExecutorSpec:
    argv: list
    shell: bool = False
    cwd: str | None = None

def execute(spec: ExecutorSpec, run_dir: Path, timeout: int, env: dict | None = None) -> dict:
    start = time.monotonic()
    with open(run_dir / "stdout.log", "w", encoding="utf-8", errors="replace") as fo, \
         open(run_dir / "stderr.log", "w", encoding="utf-8", errors="replace") as fe:
        proc = subprocess.Popen(spec.argv, shell=spec.shell, cwd=spec.cwd,
                                stdout=fo, stderr=fe, env=env)
        try:
            code = proc.wait(timeout=timeout)
            status = "success" if code == 0 else "failed"
        except subprocess.TimeoutExpired:
            subprocess.run(["taskkill", "/T", "/F", "/PID", str(proc.pid)],
                           capture_output=True)
            proc.wait()
            code, status = proc.returncode, "timeout"
    return {"exit_code": code, "duration_sec": round(time.monotonic() - start, 3),
            "status": status}
```

```python
# runner/cron_runner/executors/opencode.py
import json
from cron_runner.config import interpolate
from cron_runner.executors.base import ExecutorSpec

def build(task, step, variables):
    argv = ["opencode", "run", "--format", "json", "--title", f"cron-{task.id}"]
    if task.model:
        argv += ["-m", task.model]
    if task.permission in ("auto", "yolo"):
        argv += ["--auto"]
    argv.append(interpolate(step.run, variables))
    return ExecutorSpec(argv=argv, shell=False,
                        cwd=interpolate(task.workdir, variables) or None)

def _find_model(obj):
    if isinstance(obj, dict):
        if "model" in obj and isinstance(obj["model"], str):
            return obj["model"]
        for v in obj.values():
            m = _find_model(v)
            if m:
                return m
    elif isinstance(obj, list):
        for v in obj:
            m = _find_model(v)
            if m:
                return m
    return None

def _collect_text(obj, acc):
    if isinstance(obj, dict):
        if isinstance(obj.get("text"), str):
            acc.append(obj["text"])
        for v in obj.values():
            _collect_text(v, acc)
    elif isinstance(obj, list):
        for v in obj:
            _collect_text(v, acc)

def parse_output(raw: str) -> dict:
    try:
        data = json.loads(raw)
        acc = []
        _collect_text(data, acc)
        return {"model": _find_model(data), "text": "\n".join(acc)[:2000] or raw[:2000]}
    except Exception:
        return {"model": None, "text": raw[:2000]}
```

```python
from datetime import datetime
from pathlib import Path

from cron_runner.config import load_config

from cron_runner.executors.opencode import parse_output
from cron_runner.tasks import load_tasks, validate_task

def _find(tasks, task_id):
    for t in tasks:
        if t.id == task_id:
            return t
    names = ", ".join(t.id for t in tasks) or "(none)"
    print(f"unknown task {task_id!r}; known: {names}")
    return None

def cmd_run(task_id: str, cron_home: Path) -> int:
    cfg = load_config(cron_home)
    tasks = load_tasks(cron_home)
    task = _find(tasks, task_id)
    if task is None:
        return 2
    errs = validate_task(task, cfg)
    if errs:
        print(f"task {task_id} invalid:\n  - " + "\n  - ".join(errs))
        return 2
    if not task.enabled:
        print(f"task {task_id} is disabled; nothing to do")
        return 0
    lock = acquire_task_lock(cron_home / f"state/locks/{task.id}.lock",
                             task.id, task.timeout)
    run_dir = new_run_dir(cron_home / "runs", task.id)
    started = datetime.now().isoformat(timespec="seconds")
    if not lock.acquired:
        write_meta(run_dir, {"task_id": task.id, "status": "skipped_lock",
                             "exit_code": None, "duration_sec": 0,
                             "executor": task.executor, "model": None,
                             "permission": task.permission, "missed": False,
                             "started_at": started})
        print(f"skip: task {task.id} already running ({lock.reason})")
        return 0
    if active_weight(cron_home / "state/active") + task.weight > cfg.max_concurrent:
        write_meta(run_dir, {"task_id": task.id, "status": "skipped_busy",
                             "exit_code": None, "duration_sec": 0,
                             "executor": task.executor, "model": None,
                             "permission": task.permission, "missed": False,
                             "started_at": started})
        print(f"skip: concurrency cap {cfg.max_concurrent} reached")
        return 0
    variables = dict(cfg.variables)
    variables.setdefault("WORKSPACE", "")
    variables["RUN_DIR"] = str(run_dir)
    token = register_active(cron_home / "state/active", task.id, task.weight)
    try:
        builder = get_builder(task.executor)
        spec = builder(task, task.steps[0], variables)
        env = {**os.environ, **variables}
        result = execute_wrapper(spec, run_dir, task, env, variables)
    finally:
        unregister(token)
    write_meta(run_dir, {"task_id": task.id, "status": result["status"],
                         "exit_code": result["exit_code"],
                         "duration_sec": result["duration_sec"],
                         "executor": task.executor,
                         "model": result.get("model"),
                         "permission": task.permission, "missed": False,
                         "started_at": started})
    cleanup(cron_home / "runs", cfg.retention_days)
    return 0 if result["status"] == "success" else 1

def execute_wrapper(spec, run_dir, task, env, variables):
    from cron_runner.executors.base import execute
    result = execute(spec, run_dir, task.timeout, env=env)
    if task.executor == "opencode":
        raw = (run_dir / "stdout.log").read_text(encoding="utf-8", errors="replace")
        parsed = parse_output(raw)
        result["model"] = parsed.get("model")
        (run_dir / "summary.txt").write_text(parsed.get("text", ""),
                                             encoding="utf-8")
    return result

def cmd_list(cron_home: Path, fmt: str = "table") -> None:
    cfg = load_config(cron_home)
    tasks = load_tasks(cron_home)
    now = datetime.now()
    rows = []
    for t in tasks:
        last = latest_run(cron_home / "runs", t.id) or {}
        rows.append({
            "id": t.id, "schedule": t.schedule, "enabled": t.enabled,
            "last_status": last.get("status", "-"),
            "last_time": last.get("finished_at", "-"),
            "next_run": next_run(t.schedule, now).strftime("%Y-%m-%d %H:%M")
                        if t.enabled else "-",
        })
    if fmt == "json":
        print(json.dumps(rows, ensure_ascii=False, indent=2))
        return
    for r in rows:
        print(f"{r['id']:<20} {r['schedule']:<14} "
              f"{'on' if r['enabled'] else 'OFF':<4} "
              f"{r['last_status']:<13} {r['last_time']:<20} {r['next_run']}")

def cmd_validate(cron_home: Path) -> int:
    cfg = load_config(cron_home)
    bad = 0
    for t in load_tasks(cron_home):
        errs = validate_task(t, cfg)
        if errs:
            bad += 1
            print(f"✗ {t.id}:\n  - " + "\n  - ".join(errs))
        else:
            print(f"✓ {t.id}")
    return 1 if bad else 0

def main(argv=None):
    ap = argparse.ArgumentParser(prog="cron_runner")
    ap.add_argument("--home", default=None,
                    help="cron repo root (default: parent of this package)")
    sub = ap.add_subparsers(dest="cmd", required=True)
    p_run = sub.add_parser("run"); p_run.add_argument("task_id")
    p_list = sub.add_parser("list")
```

```python
# runner/cron_runner/scheduler_win.py(要点框架,测试驱动补全)
import subprocess, tempfile
from datetime import date
from pathlib import Path
from xml.sax.saxutils import escape

from cron_runner.config import Config
from cron_runner.tasks import Task, load_tasks, validate_task
from cron_runner.due import next_run  # noqa: F401(展示用,未用可删)

_DAYS = {0: "Sunday", 1: "Monday", 2: "Tuesday", 3: "Wednesday",
         4: "Thursday", 5: "Friday", 6: "Saturday"}
_WEEKDAY_RE = None

class UnsupportedSchedule(Exception):
    pass

def _parse(schedule: str):
    parts = schedule.split()
    if len(parts) != 5:
        raise UnsupportedSchedule(schedule)
    minute, hour, dom, mon, dow = parts
    if dom != "*" or mon != "*":
        raise UnsupportedSchedule(schedule)  # MONTHLY/复杂 → M2
    if minute.startswith("*/") and hour == "*" and dow == "*":
        return ("minute", int(minute[2:]), None)
    if "/" in minute and minute[0].isdigit() and hour == "*" and dow == "*":
        return ("minute", int(minute.split("/")[1]), int(minute.split("/")[0]))
    if minute.isdigit() and hour.isdigit():
        dow_days = _parse_dow(dow)
        return ("daily", int(hour), int(minute)) if dow_days is None \
            else ("weekly", int(hour), int(minute), dow_days)
    raise UnsupportedSchedule(schedule)

def _parse_dow(dow):
    if dow == "*":
        return None
    days = []
    for part in dow.split(","):
        if "-" in part:
            a, b = part.split("-")
            days += list(range(int(a), int(b) + 1))
        elif part.isdigit():
            days.append(int(part))
        else:
            raise UnsupportedSchedule(dow)
    return [d % 7 for d in days]

_TMPL = """<?xml version="1.0" encoding="UTF-16"?>
<Task version="1.2" xmlns="http://schemas.microsoft.com/windows/2004/02/mit/task">
  <Triggers>{triggers}</Triggers>
  <Settings>
    <MultipleInstancesPolicy>IgnoreNew</MultipleInstancesPolicy>
    <DisallowStartIfOnBatteries>false</DisallowStartIfOnBatteries>
    <StopIfGoingOnBatteries>false</StopIfGoingOnBatteries>
    <StartWhenAvailable>{swa}</StartWhenAvailable>
    <ExecutionTimeLimit>PT{limit}S</ExecutionTimeLimit>
    <Enabled>true</Enabled>
  </Settings>
  <Actions Context="Author">
    <Exec>
      <Command>{python}</Command>
      <Arguments>{args}</Arguments>
      <WorkingDirectory>{workdir}</WorkingDirectory>
    </Exec>
  </Actions>
</Task>
"""

def schtasks_xml(task: Task, cfg: Config) -> str:
    parsed = _parse(task.schedule)
    today = date.today().isoformat()
    if parsed[0] == "daily":
        _, h, m = parsed
        triggers = (f'<CalendarTrigger><StartBoundary>{today}T{h:02d}:{m:02d}:00'
                    f'</StartBoundary><Enabled>true</Enabled>'
                    f'<ScheduleByDay><DaysOfMonth><Day>1</Day></DaysOfMonth>'
                    f'</ScheduleByDay></CalendarTrigger>')
        # 注:Daily 用 ScheduleByDay 时应每月每天;上面 DaysOfMonth 写法不通用,
        # 实现时改为 <ScheduleByDay><DaysOfWeek>...</DaysOfWeek></ScheduleByDay>
        # 仅当 weekly;纯 daily 用 repetition P1D 于 CalendarTrigger。
    # ……(weekly/minute 分支按测试 golden 补全;实现必须让测试通过,不逐字照抄注释)
    swa = "true" if task.miss_policy == "coalesce" else "false"
    home = Path(cfg.variables["CRON_HOME"])
    entry = home / "runner" / "cron_runner" / "main.py"
    args = escape(f'"{entry}" --home "{home}" run {task.id}')
    return _TMPL.format(
        triggers=triggers, swa=swa, limit=task.timeout + 300,
        python=escape(__import__("sys").executable), args=args,
        workdir=escape(str(home)))
```

**实现注意(写给执行者)**:上面 `_TMPL` 与 `_parse` 是骨架,触发器 XML 三个分支(daily=CalendarTrigger+每日重复或 ScheduleByDay、weekly=Weekly+DaysOfWeek、minute=TimeTrigger+Repetition)必须以测试 golden 为准补全;`encoding="UTF-16"` 声明要与实际写盘编码一致(schtasks 要 UTF-16 文件,`Path.write_text(..., encoding="utf-16")`)。daily 触发器推荐 `<CalendarTrigger><StartBoundary>…</StartBoundary><ScheduleByDay><DaysOfWeek><Monday/>…全部七天</DaysOfWeek></ScheduleByDay></CalendarTrigger>` 或 TimeTrigger+Repetition P1D,选一种让测试与真实 schtasks 都接受的形式,并在验收任务里真注册验证。

```python
class SchtasksBackend:
    def create(self, name, xml_text):
        with tempfile.NamedTemporaryFile("w", suffix=".xml", delete=False,
                                         encoding="utf-16") as f:
            f.write(xml_text); path = f.name
        r = subprocess.run(["schtasks", "/Create", "/F", "/XML", path,
                            "/TN", name], capture_output=True, text=True)
        Path(path).unlink(missing_ok=True)
        if r.returncode != 0:
            raise RuntimeError(f"schtasks create {name} failed: {r.stderr}")
    def delete(self, name):
        r = subprocess.run(["schtasks", "/Delete", "/F", "/TN", name],
                           capture_output=True, text=True)
        return r.returncode == 0
    def names(self):
        r = subprocess.run(["schtasks", "/Query", "/FO", "CSV"],
                           capture_output=True, text=True)
        out = []
        for line in r.stdout.splitlines()[1:]:
            if line.startswith('"cron-'):
                out.append(line.split('","')[0].lstrip('"'))
        return set(out)

def cmd_sync(cron_home: Path, backend=None) -> int:
    import hashlib
    cfg = load_config(cron_home)
    be = backend or SchtasksBackend()
    state_dir = cron_home / "state" / "sync"
    state_dir.mkdir(parents=True, exist_ok=True)
    want = {}
    for t in load_tasks(cron_home):
        if not t.enabled or validate_task(t, cfg):
            continue
        try:
            want[f"cron-{t.id}"] = schtasks_xml(t, cfg)
        except UnsupportedSchedule as e:
            print(f"skip {t.id}: schedule {e} not supported until M2")
    have = be.names()
    for name in sorted(have - set(want)):
        be.delete(name)
        (state_dir / f"{name}.sha256").unlink(missing_ok=True)
        print(f"deleted {name}")
    for name, xml in sorted(want.items()):
        digest = hashlib.sha256(xml.encode("utf-8")).hexdigest()
        f = state_dir / f"{name}.sha256"
        unchanged = name in have and f.exists() and f.read_text() == digest
        if unchanged:
            continue  # 幂等:内容未变不重建(schtasks 重建会重置运行历史)
        if name in have:
            be.delete(name)
        be.create(name, xml)
        f.write_text(digest, encoding="utf-8")
        print(f"{'updated' if name in have else 'created'} {name}")
    return 0
```
